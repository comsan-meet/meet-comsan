ตั้งค่าสะพานเชื่อม Google Apps Script ฟรี เพื่อให้เว็บอ่านชื่อรูปในโฟลเดอร์ Drive ได้ทันที

เข้าเว็บไซต์ script.google.com (ใช้ Gmail หรือ KKU Mail) > กดปุ่ม "โครงการใหม่ (New project)"

วางโค้ดสคริปต์เชื่อมต่อ Drive ด้านล่างนี้

function doGet(e) {
  var folderId = e.parameter.folderId;
  if (!folderId && e.parameter.url) {
    var match = e.parameter.url.match(/folders\/([a-zA-Z0-9_-]+)/);
    if (match) folderId = match[1];
  }
  if (!folderId) {
    return ContentService.createTextOutput(JSON.stringify({
      ok: false,
      status: "error",
      message: "Missing folderId parameter"
    })).setMimeType(ContentService.MimeType.JSON);
  }
  try {
    var folder = DriveApp.getFolderById(folderId);
    var files = folder.getFiles();
    var list = [];
    while (files.hasNext()) {
      var file = files.next();
      var name = file.getName();
      var id = file.getId();
      var mime = file.getMimeType();
      if (mime.indexOf("image/") === 0 || name.match(/\.(jpg|jpeg|png|webp|gif|svg)$/i)) {
        list.push({
          id: id,
          name: name,
          size: file.getSize(),
          url: "https://lh3.googleusercontent.com/d/" + id
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({
      ok: true,
      status: "success",
      folderName: folder.getName(),
      totalFiles: list.length,
      files: list
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      ok: false,
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
